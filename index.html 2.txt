            <!-- Tab Content 1: Academic Derivation -->
            <div id="tabContent-academic-deriv" class="p-6 md:p-8 space-y-6">
                <div class="prose max-w-none text-slate-600 text-sm leading-relaxed space-y-4">
                    <h3 class="text-lg font-bold text-slate-900">Step-by-Step Analytical Derivation</h3>
                    <p>
                        To model the biological weight gain of the Harumanis mango, we apply a classic three-parameter <strong>Logistic Model</strong>. This is highly standard in forestry and crop sciences because growth exhibits early exponential behavior (unbounded cell division), followed by a linear-ish inflection phase, and finally plateaus due to spatial and biological nutrient limits (maturation).
                    </p>
                    
                    <div class="bg-slate-50 p-5 rounded-xl border border-slate-100 font-mono text-xs">
                        <strong class="text-slate-800">Mathematical Setup:</strong><br><br>
                        Let growth function be: <br>
                        $$f(x) = \frac{L}{1 + e^{-u(x)}} \quad \text{where} \quad u(x) = k(x - x_0)$$
                        <br>
                        Applying the Chain Rule and Reciprocal Rule: <br>
                        $$f(x) = L \cdot [1 + e^{-k(x - x_0)}]^{-1}$$
                        <br>
                        $$\frac{d}{dx}[f(x)] = -L \cdot [1 + e^{-k(x - x_0)}]^{-2} \cdot \frac{d}{dx}[1 + e^{-k(x - x_0)}]$$
                        <br>
                        $$\frac{d}{dx}[1 + e^{-k(x - x_0)}] = e^{-k(x - x_0)} \cdot (-k) = -k e^{-k(x - x_0)}$$
                        <br>
                        Substitute back to find the first derivative \( f'(x) \): <br>
                        $$f'(x) = -L \cdot [1 + e^{-k(x - x_0)}]^{-2} \cdot (-k e^{-k(x - x_0)}) = \frac{L \cdot k \cdot e^{-k(x - x_0)}}{(1 + e^{-k(x - x_0)})^2}$$
                    </div>

                    <h4 class="font-bold text-slate-900">Why the Inflection Point Matters:</h4>
                    <p>
                        The second derivative \( f''(x) \) shows the rate of acceleration. At \( x = 45 \) (the inflection point), the acceleration drops to zero \( (f''(x) = 0) \). This marks the exact transition from <strong>increasing rate of growth</strong> to a <strong>decreasing rate of growth</strong>, representing peak physiological stress for agricultural vascular systems.
                    </p>
                </div>
            </div>

        // Mathematical Constants based on real Harumanis crop profiles
        const L = 500;   // Maximum mature weight (grams)
        const k = 0.09;  // Growth speed coefficient
        const x0 = 45;   // Inflection day (Days After Anthesis)

        // Function f(x) - Logistic Growth Function
        function f(x) {
            return L / (1 + Math.exp(-k * (x - x0)));
        }

        // Derivative f'(x) - Rate of change function
        function df(x) {
            const expTerm = Math.exp(-k * (x - x0));
            return (L * k * expTerm) / Math.pow(1 + expTerm, 2);
        }

        // Custom interactive narrative engine for laymen/students
        function getLaymanInterpretation(day, weight, rate) {
            if (day < 15) {
                return `**Stage 1: Early Lag Phase (Rapid Cell Division)**  
                At Day ${day}, the mango weighs only **${weight.toFixed(1)}g**. The derivative is very low (**${rate.toFixed(2)} g/day**). The mango tree is dedicating resources to internal cell divisions rather than expansion. Keep watering minimal.`;
            } else if (day >= 15 && day < 35) {
                return `**Stage 2: Acceleration Phase (Vascular Expansion)**  
                At Day ${day}, growth is picking up speed (**${rate.toFixed(2)} g/day**), and weight has climbed to **${weight.toFixed(1)}g**. The cellular structure is stretching. SFA system should start scaling up water supply.`;
            } else if (day >= 35 && day <= 55) {
                const maxWarn = (day > 42 && day < 48) ? "⚠️ **CRITICAL MAXIMUM RATE:** This is the peak growth inflection stage!" : "";
                return `**Stage 3: Exponential / Log Phase (Rapid Pulp Accumulation)**  
                ${maxWarn} At Day ${day}, the fruit is gaining mass at its absolute fastest speed of **${rate.toFixed(2)} grams per day**. Cumulative weight is **${weight.toFixed(1)}g**. Any biological water stress right now will cause permanent stunting. IoT smart-sprinklers must supply maximum target irrigation.`;
            } else if (day > 55 && day < 75) {
                return `**Stage 4: Deceleration / Sugar Storing Phase**  
                At Day ${day}, physical growth speed is slowing down to **${rate.toFixed(2)} g/day**, though total weight is still rising to **${weight.toFixed(1)}g**. The mango is transitioning from physical scaling to internal sugar, organic acid, and starch conversion.`;
            } else {
                return `**Stage 5: Saturation / Physiological Ripening (Harvest Window)**  
                At Day ${day}, the derivative is nearly flat (**${rate.toFixed(2)} g/day**) as weight reaches **${weight.toFixed(1)}g** of its ${L}g target. Growth rate is zero, meaning physical growth has ceased. It is ready for picking! SFA algorithms flag this stabilization to schedule human harvesting crews.`;
            }
        }

        // Generate dataset coordinates
        const daysArray = [];
        const massData = [];
        const rateData = [];

        for (let x = 0; x <= 90; x++) {
            daysArray.push(x);
            massData.push(f(x));
            rateData.push(df(x) * 15); // scaled up by factor of 15 for visualization clarity on the double axis
        }

        let activeX = 45; // Start centered at the Inflection Point
        let growthChart;

        // Custom Tangent line generator helper
        function getTangentLineDataset(xVal) {
            const yVal = f(xVal);
            const slope = df(xVal);
            const tangentPoints = [];
            
            // Generate a small tangent segment extending +/- 15 days around active point
            const range = 15;
            const startX = Math.max(0, xVal - range);
            const endX = Math.min(90, xVal + range);
            
            for (let x = 0; x <= 90; x++) {
                if (x >= startX && x <= endX) {
                    // Equation of tangent line: y = m(x - x1) + y1
                    tangentPoints.push({
                        x: x,
                        y: slope * (x - xVal) + yVal
                    });
                } else {
                    tangentPoints.push({ x: x, y: null });
                }
            }
            return tangentPoints;
        }

        const gradEdge = document.getElementById('gradEdge');

        // Primary application update dispatcher
        function updateSimulation(dayVal) {
            activeX = parseInt(dayVal);
            
            // 1. Compute values
            const weight = f(activeX);
            const rate = df(activeX);

            // 2. Update text & numbers
            badge.innerText = `Day ${activeX} DAA`;
            metricDay.innerText = activeX;
            metricWeight.innerText = weight.toFixed(1);
            metricDerivative.innerText = `+${rate.toFixed(2)}`;
            slider.value = activeX;

            // Render Markdown-styled bold/italic text nicely
            const explanation = getLaymanInterpretation(activeX, weight, rate);
            explanationText.innerHTML = explanation
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
                .replace(/\*(.*?)\*/g, '<em>$1</em>');

            // 3. Update Visual Mango Avatar Sizing & Ripeness Color Transition
            // Sizing scaling based on physical fruit weight
            const scaleFactor = 0.4 + (weight / L) * 0.75; // dynamic scaling from 0.4x to 1.15x
            mangoSvg.style.transform = `scale(${scaleFactor})`;
            
            // Leaf rotates dynamically as weight bends the stem
            const leafAngle = 15 + (activeX / 90) * 45;
            mangoLeaf.style.transform = `rotate(${leafAngle}deg)`;

            // Color transition from early bright-green to ripe golden Harumanis yellow-green
            if (activeX < 35) {

            }

            // 4. Update dynamic tangent line and point annotation dataset on chart

                // Update tangent line
                growthChart.data.datasets[2].data = getTangentLineDataset(activeX);

                growthChart.data.datasets[3].data = [{ x: activeX, y: weight }];
                growthChart.update('none'); // silent update without reset animations
            }
